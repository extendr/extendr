64fn collect_vector<T, U>(
65    sexptype: SEXPTYPE,
66    data: impl Fn(SEXP) -> *mut U,
67    iter: impl Iterator<Item = T>,
68    convert: impl Fn(T) -> U,
69) -> Robj {
70    unsafe {
71        let mut capacity = iter.size_hint().0;
72        let mut robj = Robj::alloc_vector(sexptype, capacity);
73        let mut ptr = data(robj.get());
74        let mut n = 0;
75        for v in iter {
76            if n == capacity {
77                capacity = (capacity * 2).max(1);
78                robj = Robj::from_sexp(Rf_xlengthgets(robj.get(), capacity as R_xlen_t));
79                ptr = data(robj.get());
80            }
81            ptr.add(n).write(convert(v));
82            n += 1;
83        }
84        if n < capacity {
85            robj = Robj::from_sexp(Rf_xlengthgets(robj.get(), n as R_xlen_t));
86        }
87        robj
88    }
